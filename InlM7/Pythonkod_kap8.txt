def är_perfekt(n):
    if n <= 1: # 1 eller mindre: inte perfekt
        return False

    summa_delare = 0
    for i in range(1, n // 2 + 1): # kollar till hälften
        if n % i == 0: 
            summa_delare += i

    return summa_delare == n

def störst(sekvens):
    if not sekvens:
        return None
    if len(sekvens) == 1:
        return sekvens[0]

    max_resten = störst(sekvens[1:])
    return sekvens[0] if sekvens[0] > max_resten else max_resten

def pris_ink_moms(pris_exk, momssats):
    # Räknar om momssatsen till multiplikationsfaktor: t.ex. 25% -> 1.25
    return pris_exk * (1 + momssats / 100)

def räkna_bokstäver(text):
    små = 0
    stora = 0
    for c in text:
        if c.islower():
            små += 1
        elif c.isupper():
            stora += 1
    # Returnerar paret i en tupel
    return (små, stora)


# Test
print(räkna_bokstäver("Hej Världen!"))  # (8, 2)

def primtal(n):
    if n <= 1:
        return False
    # Om något tal mellan 2 och n-1 delar n jämnt är det inte ett primtal
    for i in range(2, n):
        if n % i == 0:
            return False
    return True


# Test
print(primtal(7))  # True
print(primtal(4))  # False

def nfak(n):
    # Beräknar n! (n-fakultet)
    res = 1
    for i in range(1, n + 1):
        res *= i
    return res


def binomialkoefficient(n, k):
    # Formeln: n! / (k! * (n - k)!)
    return nfak(n) // (nfak(k) * nfak(n - k))


# Test
print(binomialkoefficient(5, 2))  # 10

def kvadratrot(x):
    g = x / 2  # Första gissningen
    while True:
        # Newtons formel: medelvärdet av g och x/g
        g_ny = (g + x / g) / 2
        # Avbryt när förändringen är mindre än 10^-6
        if abs(g_ny - g) < 1e-6:
            return g_ny
        g = g_ny


# Test
print(kvadratrot(9))  # ~3.0
print(kvadratrot(2))  # ~1.41421356
